```go
}
```

```go
type Goal struct {
	ID                 uuid.UUID
	TenantID           uuid.UUID
	Name               string
	GoalType           GoalType
	TargetAmountCents  int64
	CurrentAmountCents int64
	CurrencyCode       string
	Deadline           *time.Time
	LinkedAccountIDs   []uuid.UUID // 替换 LinkedAccountID(Investment + Savings)
	LinkedDebtIDs      []uuid.UUID // DebtPayoff
	Notes              string
	IsCompleted        bool
	Version            int64
	CreatedAt          time.Time
	UpdatedAt          time.Time
}

func NewGoal(tenantID uuid.UUID, name string, goalType GoalType, targetAmountCents int64, currencyCode string, deadline *time.Time, linkedAccountIDs []uuid.UUID, linkedDebtIDs []uuid.UUID, notes string) (*Goal, error) {
	name = trimSpace(name)
	if name == "" {
		return nil, fmt.Errorf("goal name must not be empty")
	}
	if targetAmountCents <= 0 {
		return nil, fmt.Errorf("target amount must be positive")
	}
	if goalType == 0 {
		return nil, fmt.Errorf("goal type must be specified")
	}
	if currencyCode == "" {
		currencyCode = "CNY"
	}
	// 按 type 校验关联
	switch goalType {
	case GoalTypeInvestment, GoalTypeSavings:
		if len(linkedAccountIDs) == 0 {
			return nil, fmt.Errorf("%s goal requires at least 1 linked account", goalType)
		}
	case GoalTypeDebtPayoff:
		if len(linkedDebtIDs) == 0 {
			return nil, fmt.Errorf("debtpayoff goal requires at least 1 linked debt")
		}
	}
	now := time.Now()
	return &Goal{
		ID: uuid.New(), TenantID: tenantID, Name: name, GoalType: goalType,
		TargetAmountCents: targetAmountCents, CurrentAmountCents: 0, CurrencyCode: currencyCode,
		Deadline: deadline, LinkedAccountIDs: linkedAccountIDs, LinkedDebtIDs: linkedDebtIDs,
		Notes: notes, Version: 1, CreatedAt: now, UpdatedAt: now,
	}, nil
}

// Clone 复制 goal(可改 target/deadline/name),reset current + 新 ID。关联深拷。
func (g *Goal) Clone(newTenantID uuid.UUID, targetAmountCents int64, deadline *time.Time, name string) (*Goal, error) {
	if name == "" {
		name = g.Name
	}
	if targetAmountCents <= 0 {
		targetAmountCents = g.TargetAmountCents
	}
	accs := append([]uuid.UUID(nil), g.LinkedAccountIDs...)
	debts := append([]uuid.UUID(nil), g.LinkedDebtIDs...)
	return NewGoal(newTenantID, name, g.GoalType, targetAmountCents, g.CurrencyCode, deadline, accs, debts, g.Notes)
}

// SetCurrentAmount(scheduler 算 progress 用, D-goal 已有, 保留)
func (g *Goal) SetCurrentAmount(amtCents int64) { /* D-goal 既有, 不改 */ }
```

```go
func (s *Service) GetAccountsBalance(ctx context.Context, tenantID uuid.UUID, accountIDs []uuid.UUID) (int64, error) {
```

```go
// application/service.go
type Service struct {
	repo      domain.GoalRepository
	mvSrc     domain.AccountMarketValueSource // nil = Investment goal skip
	balSrc    domain.AccountBalanceSource     // nil = Savings goal skip
	debtSrc   domain.DebtProgressSource       // nil = DebtPayoff goal skip
	// ... D-goal 既有字段
}

func (s *Service) SetAccountMarketValueSource(src domain.AccountMarketValueSource)  { s.mvSrc = src }
func (s *Service) SetAccountBalanceSource(src domain.AccountBalanceSource)          { s.balSrc = src }
func (s *Service) SetDebtProgressSource(src domain.DebtProgressSource)              { s.debtSrc = src }

// SyncAllGoals recomputes all goals' current_amount by type (Investment mv /
// Savings balance / DebtPayoff paid) + writes daily snapshot. best-effort:
// per-goal/port fail → skip + log. 替 D-goal SyncInvestmentGoals。
func (s *Service) SyncAllGoals(ctx context.Context, tenantID uuid.UUID) (int, error) {
	goals, err := s.repo.FindAll(ctx, tenantID) // 照 D-goal 现有 FindAll
	if err != nil {
		return 0, fmt.Errorf("list goals: %w", err)
	}
	count := 0
	for _, g := range goals {
		var cur int64
		switch g.GoalType {
		case domain.GoalTypeInvestment:
			if s.mvSrc == nil || len(g.LinkedAccountIDs) == 0 {
				continue
			}
			cur, err = s.mvSrc.GetAccountsMarketValue(ctx, tenantID, g.LinkedAccountIDs)
		case domain.GoalTypeSavings:
			if s.balSrc == nil || len(g.LinkedAccountIDs) == 0 {
				continue
			}
			cur, err = s.balSrc.GetAccountsBalance(ctx, tenantID, g.LinkedAccountIDs)
		case domain.GoalTypeDebtPayoff:
			if s.debtSrc == nil || len(g.LinkedDebtIDs) == 0 {
				continue
			}
			cur, err = s.debtSrc.GetDebtsPaid(ctx, tenantID, g.LinkedDebtIDs)
		default:
			continue
		}
		if err != nil {
			slog.Error("goal sync: port failed", "goal_id", g.ID.String(), "type", g.GoalType.String(), "error", err.Error())
			continue
		}
		g.SetCurrentAmount(cur)
		if err := s.repo.Update(ctx, g); err != nil { // 写 current(含多账户 links)
			slog.Error("goal sync: update failed", "goal_id", g.ID.String(), "error", err.Error())
			continue
		}
		if err := s.repo.WriteSnapshot(ctx, g); err != nil { // snapshot upsert(照 holding_snapshot)
			slog.Warn("goal sync: snapshot write failed", "goal_id", g.ID.String(), "error", err.Error())
		}
		count++
	}
	return count, nil
}
```
