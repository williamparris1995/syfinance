```dart
Widget _subline(Account a) {
  switch (a.category) {
    case AccountCategory.credit:
      return Text('额度 ${fmt(a.creditLimitCents)} · 账单${a.creditBillingDay}日 / 还款${a.creditRepaymentDay}日', ...);
    case AccountCategory.invest:
      final r = a.investReturnYtd;
      return Text('${r>=0?'+':''}${r.toStringAsFixed(2)}% 今年收益', style: TextStyle(color: r>=0?AppColors.positive:AppColors.negative));
    case AccountCategory.fixed:
      return Text('到期 ${fmtDate(a.fixedMaturityDate)} · 利率 ${a.interestRate?.toStringAsFixed(2)}%');
    case AccountCategory.gold:
      final now = (a.goldCurrentPriceCents * (a.goldQuantity ?? 0)).toInt();
      final pct = a.goldBuyPriceCents > 0 ? (now - a.goldBuyPriceCents) / a.goldBuyPriceCents * 100 : 0.0;
      return Text('现值 ${fmt(now)} · 买入 ${fmt(a.goldBuyPriceCents)} · ${pct>=0?'+':''}${pct.toStringAsFixed(2)}%');
    case AccountCategory.estate:
      final pct = a.estatePurchasePriceCents > 0 ? (a.estateCurrentValueCents - a.estatePurchasePriceCents) / a.estatePurchasePriceCents * 100 : 0.0;
      return Text('现估值 ${fmt(a.estateCurrentValueCents)} · ${pct>=0?'+':''}${pct.toStringAsFixed(2)}%');
    case AccountCategory.loan:
      return Text('原始 ${fmt(a.loanOriginalCents)} · 月供 ${fmt(a.loanMonthlyCents)} · 下次 ${fmtDate(a.loanNextPaymentDate)}');
    default: // savings/现金
      return a.interestRate != null ? Text('利率 ${a.interestRate!.toStringAsFixed(2)}%') : SizedBox.shrink();
  }
}
```

```dart
Widget _hero(Account a) {
  final txnState = context.watch<TransactionBloc>().state;
  final net = (txnState is TransactionsLoaded ? txnState.summary?.netCents : null) ?? 0;
  return ClipRRect(
    borderRadius: AppRadius.lgBorder,
    child: Container(
      padding: EdgeInsets.all(32),
      decoration: BoxDecoration(gradient: LinearGradient(colors:[Color(0xFF1C1E21), Color(0xFF2A2D33)])),
      child: Stack(children: [
        Positioned(top:-40, right:-40, child: Container(width:180,height:180, decoration: BoxDecoration(shape:BoxShape.circle, gradient: RadialGradient(colors:[AppColors.accent.withValues(alpha:0.18), Colors.transparent])))),
        Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
          // hero-badge: 类型 + 资产·负债 + 活期/定期
          Wrap(spacing:8, children: [
            _heroBadge(a.category.label),
            _heroBadge(a.accountType == AccountType.liability ? '负债类' : '资产类'),
            if (a.category == AccountCategory.fixed) _heroBadge('定期') else _heroBadge('活期'),
          ]),
          SizedBox(height:12),
          // 余额 40px 白字 serif
          Text(fmt(a.currentBalanceCents), style: TextStyle(fontSize:40, fontWeight:FontWeight.w600, color:Colors.white, fontFamily:AppTypography.displayFamily)),
          SizedBox(height:8),
          // 本月收支副信息（net 正绿负红）
          Text('本月收支 ${net>=0?'+':'-'}¥${(net.abs()/100).toStringAsFixed(2)}', style: TextStyle(color: net>=0?Color(0xFF6FCF9A):Color(0xFFE57373))),
          SizedBox(height:20),
          // 字段网格（类型专属，desktop 4 列 / mobile 2 列）
          _heroFields(a),
        ]),
      ]),
    ),
  );
}

Widget _heroFields(Account a) {
  final fields = <(String,String)>[]; // (label, value)
  switch (a.category) {
    case AccountCategory.credit: fields.addAll([ ('额度',fmt(a.creditLimitCents)), ('账单日','${a.creditBillingDay}日'), ('还款日','${a.creditRepaymentDay}日'), ('年费',fmt(a.creditAnnualFeeCents)) ]);
    case AccountCategory.loan: fields.addAll([ ('原始本金',fmt(a.loanOriginalCents)), ('剩余本金',fmt(a.loanRemainingCents)), ('月供',fmt(a.loanMonthlyCents)), ('下次还款',fmtDate(a.loanNextPaymentDate)) ]);
    case AccountCategory.invest: fields.addAll([ ('市值',fmt(a.investMarketValueCents)), ('成本',fmt(a.investCostCents)), ('今年收益率','${a.investReturnYtd?.toStringAsFixed(2)}%'), if(a.investMarketValueCents>0) ('持仓数','—') ]);
    case AccountCategory.gold: fields.addAll([ ('品种',a.goldProductType), ('数量','${a.goldQuantity?.toStringAsFixed(3)}'), ('买入价',fmt(a.goldBuyPriceCents)), ('现价',fmt(a.goldCurrentPriceCents)) ]);
    case AccountCategory.estate: fields.addAll([ ('买入价',fmt(a.estatePurchasePriceCents)), ('现估值',fmt(a.estateCurrentValueCents)), ('买入日期',fmtDate(a.estatePurchaseDate)), ('折旧率','${a.estateDepreciationRate?.toStringAsFixed(2)}%') ]);
    case AccountCategory.fixed: fields.addAll([ ('本金',fmt(a.fixedPrincipalCents)), ('起息日',fmtDate(a.fixedStartDate)), ('到期日',fmtDate(a.fixedMaturityDate)), ('期限','${a.fixedTermMonths}月') ]);
    default: fields.addAll([ if(a.interestRate!=null) ('利率','${a.interestRate!.toStringAsFixed(2)}%'), ('开户日期', fmtDate(a.openingDate)), ('币种',a.currencyCode) ]);
  }
  return LayoutBuilder(builder:(ctx,c)=> GridView.count(
    shrinkWrap:true, physics:NeverScrollableScrollPhysics(),
    crossAxisCount: c.maxWidth > 600 ? 4 : 2,
    mainAxisSpacing:12, crossAxisSpacing:12, childAspectRatio: 2.5,
    children: [for(final f in fields) _heroField(f.$1, f.$2)],
  ));
}
```

TxnRow 加 `Map<String,Account> accounts` 参数（若缺）。账户列渲染：
```dart
// _TxCard 账户列 / _MobileTxnCard 副行
final isTransfer = txn.entries.length == 2 && txn.entries.every((e) => /* 两 asset account */);
if (isTransfer) {
  final from = txn.entries.firstWhere((e) => e.creditCents > 0);  // 贷方=转出
  final to = txn.entries.firstWhere((e) => e.debitCents > 0);     // 借方=转入
  final fromAcct = accounts[from.accountId];
  final toAcct = accounts[to.accountId];
  return Row(children: [
    _acctChip(fromAcct),  // 首字母方块 + 名
    Icon(Icons.arrow_forward, size:14, color: AppColors.muted),
    _acctChip(toAcct),
  ]);
} else {
  // 非转账：单资产账户 + 分类 chip（首条非 asset entry 的账户 = 分类）
  final assetEntry = txn.entries.firstWhere((e) => accounts[e.accountId]?.accountType == AccountType.asset);
  final catEntry = txn.entries.firstWhere((e) => accounts[e.accountId]?.accountType != AccountType.asset);
  return Row(children: [ _acctChip(accounts[assetEntry.accountId]), _categoryChip(accounts[catEntry.accountId]) ]);
}
```

`_acctChip(Account a)`：首字母方块（account 类型色）+ 名。`_categoryChip`：分类 chip（收入绿/支出红）。

transactions_page 传 accounts 缓存（已为 FilterBar 加载）给 TxnRow。
